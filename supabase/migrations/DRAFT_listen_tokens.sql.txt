-- Listen tokens: shareable, revocable links to a single R2 object.
-- A token is an unguessable uuid that maps to one file_key; the presign
-- endpoint (future /api/listen-url) resolves it via the SECURITY DEFINER
-- function below — anon callers can EXECUTE the function but can never
-- SELECT the table, so enumeration/scraping of tokens is impossible and
-- a revoked or expired token resolves to nothing.

create table public.listen_tokens (
  token          uuid primary key default gen_random_uuid(),
  file_key       text not null,
  created_by     uuid references auth.users(id) on delete cascade,
  expires_at     timestamptz not null default now() + interval '30 days',
  allow_download boolean not null default false,
  revoked_at     timestamptz,
  created_at     timestamptz default now()
);

alter table public.listen_tokens enable row level security;

-- Deliberately NO select policy: direct reads are deny-all, even for
-- authenticated users. The only read path is resolve_listen_token().

create policy "owners insert their own tokens"
  on public.listen_tokens for insert
  to authenticated
  with check (created_by = auth.uid());

-- Owners can revoke / extend their own tokens.
create policy "owners update their own tokens"
  on public.listen_tokens for update
  to authenticated
  using (created_by = auth.uid())
  with check (created_by = auth.uid());

-- Resolve a token to its file. SECURITY DEFINER so it can read past the
-- deny-all RLS; returns zero rows for unknown, revoked, or expired
-- tokens — the caller learns nothing beyond valid/invalid.
create or replace function public.resolve_listen_token(t uuid)
returns table (file_key text, allow_download boolean)
language sql
security definer
set search_path = public
stable
as $$
  select lt.file_key, lt.allow_download
    from public.listen_tokens lt
   where lt.token = t
     and lt.revoked_at is null
     and lt.expires_at > now();
$$;

grant execute on function public.resolve_listen_token(uuid) to anon, authenticated;
